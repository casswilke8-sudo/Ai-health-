export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') { res.status(200).end(); return; } if (req.method !== 'POST') { res.status(405).json({ error: 'Method not allowed' }); return; } const { goalWeight, healthGoals, medicalHistory, digestiveIssues, dailySchedule, mealPrep, diet, groceryBudget, calorieGoal } = req.body; const prompt = `You are a certified nutritionist and personal trainer. Create a personalized wellness plan. CRITICAL RULES: - Any allergy listed: NEVER include that food anywhere - Weight loss: low-calorie, high-fiber, high-protein foods - Muscle gain: high-protein foods, 2500-3000 calories - Vegan: no meat/fish/dairy/eggs. Vegetarian: no meat/fish - Bloating issues: avoid raw onions, excess beans, carbonated drinks - Match meals to their prep routine and budget Person's profile: Goal: ${goalWeight} Health goals: ${healthGoals} Allergies/medical: ${medicalHistory || 'None'} Digestive issues: ${digestiveIssues || 'None'} Schedule: ${dailySchedule || 'Not specified'} Meal prep: ${mealPrep || 'Not specified'} Diet: ${diet || 'No restrictions'} Budget: ${groceryBudget || 'Not specified'} Daily calorie goal: ${calorieGoal || 'standard for their goals'} YOU MUST respond using ONLY this exact structure. Each day and each meal MUST be on its own separate line. No exceptions. MEAL_START MONDAY Breakfast: [specific food] Lunch: [specific food] Dinner: [specific food] Snack: [specific food] TUESDAY Breakfast: [specific food] Lunch: [specific food] Dinner: [specific food] Snack: [specific food] WEDNESDAY Breakfast: [specific food] Lunch: [specific food] Dinner: [specific food] Snack: [specific food] THURSDAY Breakfast: [specific food] Lunch: [specific food] Dinner: [specific food] Snack: [specific food] FRIDAY Breakfast: [specific food] Lunch: [specific food] Dinner: [specific food] Snack: [specific food] SATURDAY Breakfast: [specific food] Lunch: [specific food] Dinner: [specific food] Snack: [specific food] SUNDAY Breakfast: [specific food] Lunch: [specific food] Dinner: [specific food] Snack: [specific food] MEAL_END WORKOUT_START MONDAY Focus: [type of workout] Exercise 1: [name] - [sets x reps or duration] Exercise 2: [name] - [sets x reps or duration] Exercise 3: [name] - [sets x reps or duration] TUESDAY Focus: [type of workout] Exercise 1: [name] - [sets x reps or duration] Exercise 2: [name] - [sets x reps or duration] WEDNESDAY Focus: [type of workout] Exercise 1: [name] - [sets x reps or duration] THURSDAY Focus: [type of workout] Exercise 1: [name] - [sets x reps or duration] Exercise 2: [name] - [sets x reps or duration] Exercise 3: [name] - [sets x reps or duration] FRIDAY Focus: [type of workout] Exercise 1: [name] - [sets x reps or duration] Exercise 2: [name] - [sets x reps or duration] SATURDAY Focus: [type of workout] Exercise 1: [name] - [sets x reps or duration] Exercise 2: [name] - [sets x reps or duration] SUNDAY Focus: Rest or light activity WORKOUT_END TIPS_START NUTRITION Tip: [specific tip for their goals] Tip: [specific tip for their goals] DIGESTION Tip: [specific tip for their digestive issues] Tip: [specific tip] FITNESS Tip: [specific tip for their schedule] Tip: [specific tip] MINDSET Tip: [specific tip] Tip: [specific tip] TIPS_END`; try { const response = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': process.env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' }, body: JSON.stringify({ model: 'claude-haiku-4-5-20251001', max_tokens: 4000, messages: [{ role: 'user', content: prompt }] }) }); const data = await response.json(); if (!response.ok) throw new Error(data.error?.message || 'API error'); const text = data.content?.[0]?.text; if (!text) throw new Error('No response from AI'); // Extract sections const mealMatch = text.match(/MEAL_START([\s\S]*?)MEAL_END/); const workoutMatch = text.match(/WORKOUT_START([\s\S]*?)WORKOUT_END/); const tipsMatch = text.match(/TIPS_START([\s\S]*?)TIPS_END/); // If delimiters not found, try to parse the raw text by splitting on day names let meal = mealMatch ? mealMatch[1].trim() : ''; let workout = workoutMatch ? workoutMatch[1].trim() : ''; let tips = tipsMatch ? tipsMatch[1].trim() : ''; // If AI ignored format, force-parse it by inserting line breaks before day names and meal types function forceFormat(raw) { if (!raw) return ''; const days = ['MONDAY','TUESDAY','WEDNESDAY','THURSDAY','FRIDAY','SATURDAY','SUNDAY']; const categories = ['NUTRITION','DIGESTION','FITNESS','MINDSET']; let result = raw; // Insert newline before each day name days.forEach(d => { result = result.replace(new RegExp('([.!?]\\s*)(' + d + ')', 'g'), '$1\n$2'); result = result.replace(new RegExp('(\\s)(' + d + ':', 'g'), '\n$2'); }); categories.forEach(c => { result = result.replace(new RegExp('([.!?]\\s*)(' + c + ')', 'g'), '$1\n$2'); }); // Insert newline before Breakfast/Lunch/Dinner/Snack/Exercise/Tip/Focus ['Breakfast','Lunch','Dinner','Snack','Exercise 1','Exercise 2','Exercise 3','Focus','Tip'].forEach(k => { result = result.replace(new RegExp('([.!?]\\s*)(' + k + ':', 'g'), '$1\n$2:'); }); return result.trim(); } // If sections not found, try to split the whole text if (!meal && !workout && !tips) { const formatted = forceFormat(text); meal = formatted; } res.status(200).json({ meal: meal || '', workout: workout || '', tips: tips || '' }); } catch (error) { res.status(500).json({ error: error.message }); } }
